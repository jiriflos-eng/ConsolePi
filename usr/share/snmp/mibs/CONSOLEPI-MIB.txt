IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, enterprises, Integer32, Gauge32, Counter64, TimeTicks
        FROM SNMPv2-SMI
    DisplayString
        FROM SNMPv2-TC;

    LAST-UPDATED "20260730Z"
    ORGANIZATION "ConsolePi"
    CONTACT-INFO "Local ConsolePi administrator"
    DESCRIPTION "Read-only operational metrics exported by ConsolePi."
    ::= { enterprises 55555 }

consolePiServices OBJECT IDENTIFIER ::= { consolePiObjects 20 }
consolePiServiceTable OBJECT-TYPE
    SYNTAX SEQUENCE OF ConsolePiServiceEntry
    MAX-ACCESS not-accessible
    STATUS current
    DESCRIPTION "ConsolePi service state table."
    ::= { consolePiServices 1 }
ConsolePiServiceEntry ::= SEQUENCE {
    consolePiServiceName DisplayString,
    consolePiServiceState Integer32
}
consolePiServiceEntry OBJECT-TYPE
    SYNTAX ConsolePiServiceEntry
    MAX-ACCESS not-accessible
    STATUS current
    DESCRIPTION "A ConsolePi service."
    INDEX { consolePiServiceIndex }
    ::= { consolePiServiceTable 1 }
consolePiServiceIndex OBJECT-TYPE
    SYNTAX Integer32 (1..255)
    MAX-ACCESS not-accessible
    STATUS current
    DESCRIPTION "Stable service index."
    ::= { consolePiServiceEntry 1 }
consolePiServiceName OBJECT-TYPE
    SYNTAX DisplayString
    MAX-ACCESS read-only
    STATUS current
    DESCRIPTION "systemd service name."
    ::= { consolePiServiceEntry 1 }
consolePiServiceState OBJECT-TYPE
    SYNTAX INTEGER { active(1), inactive(2) }
    MAX-ACCESS read-only
    STATUS current
    DESCRIPTION "Current systemd service state."
    ::= { consolePiServiceEntry 2 }
